import os
import time
import requests
import pandas as pd
from pathlib import Path
from playwright.sync_api import sync_playwright
from datetime import datetime, timedelta, timezone

# --- CONFIGURAZIONE PARAMETRI E PERCORSI ---
# Puntiamo alla cartella del progetto sul Desktop in modo dinamico
BASE_PROJECT_PATH = Path.home() / "Desktop" / "scrapingForFood"
IMAGE_DIR = BASE_PROJECT_PATH / "imagesFoodML"
FILE_NAME = BASE_PROJECT_PATH / "reddit_food_full_dataset.json"

SUBREDDIT = "food"
KEYWORDS = ["burger", "pasta", "sushi", "cake", "mexican"] 

# --- VINCOLI DI ESECUZIONE (SAFETY LIMITS) ---
MAX_POSTS_PER_KEYWORD = 50 
MAX_TOTAL_POSTS = 100 
MAX_RUNTIME_MINUTES = 30 

# Inizializzazione infrastruttura directory (Crea la cartella progetto e immagini)
if not IMAGE_DIR.exists():
    IMAGE_DIR.mkdir(parents=True, exist_ok=True)
    print(f"[FS] Directory inizializzata: {IMAGE_DIR}")

def download_image(url, post_id):
    """
    Esegue la persistenza degli asset multimediali su disco.
    Filtra gli URL per garantire lo scaricamento di soli file immagine validi.
    """
    try:
        # Pulizia URL da parametri e analisi estensione
        clean_url = url.split('?')[0]
        ext = clean_url.split('.')[-1].lower()
        if ext not in ['jpg', 'jpeg', 'png', 'webp']:
            ext = 'jpg'
        
        file_path = IMAGE_DIR / f"{post_id}.{ext}"
        
        response = requests.get(url, stream=True, timeout=10)
        if response.status_code == 200:
            with open(file_path, 'wb') as f:
                for chunk in response.iter_content(1024):
                    f.write(chunk)
            return str(file_path)
    except Exception as e:
        print(f"  [ERROR] Download fallito per {post_id}: {e}")
    return None

def get_current_total():
    """Restituisce il conteggio dei record nel file JSON per la gestione dei limiti globali."""
    if FILE_NAME.exists():
        try:
            temp_df = pd.read_json(FILE_NAME)
            return len(temp_df)
        except:
            return 0
    return 0

def save_and_append_data(new_posts):
    """Gestisce la persistenza dei dati con logica di merge e deduplicazione."""
    if not new_posts: return
    
    if FILE_NAME.exists():
        try:
            existing_df = pd.read_json(FILE_NAME)
        except:
            existing_df = pd.DataFrame()
    else:
        existing_df = pd.DataFrame()

    new_df = pd.DataFrame(new_posts)
    final_df = pd.concat([existing_df, new_df], ignore_index=True)
    
    if not final_df.empty:
        # Deduplicazione basata sull'URL unico del post
        final_df.drop_duplicates(subset=['url'], keep='first', inplace=True)
    
    final_df.to_json(FILE_NAME, orient="records", indent=4)
    print(f"\n[LOG] Dataset aggiornato. Record totali: {len(final_df)}")

def scrape_reddit_multi_keyword(subreddit_name):
    """Modulo principale di orchestrazione dello scraping con filtri restrittivi."""
    date_limit = datetime.now(timezone.utc) - timedelta(days=30)
    execution_start = time.time()
    
    with sync_playwright() as p:
        browser = p.chromium.launch(headless=False)
        context = browser.new_context(locale="en-US")
        page = context.new_page()

        for query in KEYWORDS:
            # Controllo limite globale post e tempo trascorso
            if get_current_total() >= MAX_TOTAL_POSTS: break
            elapsed = (time.time() - execution_start) / 60
            if elapsed >= MAX_RUNTIME_MINUTES: break

            print(f"\n--- ELABORAZIONE QUERY: {query.upper()} (Elapsed: {elapsed:.1f} min) ---")
            
            search_endpoint = f"https://www.reddit.com/r/{subreddit_name}/search/?q={query}&restrict_sr=1&t=all"
            page.goto(search_endpoint)
            time.sleep(3)
            try: page.get_by_role("button", name="Accept all", exact=False).click()
            except: pass

            # Scrolling per il caricamento dinamico dei link
            for _ in range(6):
                page.mouse.wheel(0, 3000)
                time.sleep(2)

            # Acquisizione link dei post dalla pagina di ricerca
            links = [n.get_attribute('href') for n in page.locator('a[data-testid="post-title"]').all()]
            unique_links = list(set([f"https://www.reddit.com{l}" if l.startswith('/') else l for l in links]))

            current_batch = []
            for i, post_url in enumerate(unique_links):
                # Verifica limiti durante l'iterazione profonda
                if (time.time() - execution_start) / 60 >= MAX_RUNTIME_MINUTES: break
                if len(current_batch) >= MAX_POSTS_PER_KEYWORD: break

                try:
                    page.goto(post_url)
                    page.wait_for_selector('shreddit-post', timeout=8000)
                    
                    post_object = page.locator('shreddit-post').first
                    
                    # --- FILTRO RESTRIZIONI (SOLO IMMAGINI SINGOLE) ---
                    # 1. Esclusione Gallerie, Video e Link esterni
                    post_type = post_object.get_attribute('post-type')
                    if post_type != 'image': continue

                    # 2. Validazione Sorgente (Solo server Reddit i.redd.it)
                    image_url = post_object.get_attribute('content-href')
                    if not image_url or "i.redd.it" not in image_url: continue

                    # 3. Validazione Estensione
                    valid_exts = ('.jpg', '.jpeg', '.png', '.webp')
                    if not image_url.split('?')[0].lower().endswith(valid_exts): continue

                    # Se il post supera i filtri, procediamo all'estrazione
                    post_id = post_object.get_attribute('id')
                    raw_ts = post_object.get_attribute('created-timestamp')
                    dt_object = datetime.fromisoformat(raw_ts.replace('Z', '+00:00'))

                    if dt_object < date_limit:
                        # Esecuzione persistenza immagine
                        local_path = download_image(image_url, post_id)
                        
                        if local_path:
                            record = {
                                "post_id": post_id,
                                "title": post_object.get_attribute('post-title'),
                                "score": int(post_object.get_attribute('score') or 0),
                                "author": post_object.get_attribute('author'),
                                "comments": int(post_object.get_attribute('comment-count') or 0),
                                "date_full": dt_object.strftime('%Y-%m-%d %H:%M:%S'),
                                "hour": dt_object.hour,
                                "day_of_week": dt_object.strftime('%A'),
                                "is_weekend": dt_object.weekday() >= 5,
                                "url": post_url,
                                "image_url": image_url,
                                "local_path": local_path,
                                "query_used": query
                            }
                            current_batch.append(record)
                            print(f"  [{len(current_batch)}] Validato e scaricato: {post_id}")
                    
                    time.sleep(1)
                except Exception: continue
            
            # Persistenza batch per keyword
            if current_batch:
                save_and_append_data(current_batch)

        browser.close()

if __name__ == "__main__":
    scrape_reddit_multi_keyword(SUBREDDIT)
